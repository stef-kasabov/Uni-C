Оператор if(){} в C

Синтаксис:
if( /*Условие*/ )
{
    //Код за въвеждане на условие
}

Пример:
#include <stdio.h>
int main() {
    int a;
    printf("Въведете число: ");
    scanf("%d", &a);
    
    if (a > 0) {
        printf("Числото е положително.\n");
    }
    
    return 0;
}

Оператор if(){} else{} в C
Синтаксис:
if( /*Условие*/ )
{
    //Код за въвеждане на условие
}
else
{
    //Код за въвеждане на условие, ако първото условие не е изпълнено
}

Пример:
#include <stdio.h>
int main() 
{
    int a;
    printf("Въведете число: ");
    scanf("%d", &a);
    
    if (a > 0) {
        printf("Числото е положително.\n");
    } else {
        printf("Числото не е положително.\n");
    }
    
    return 0;
}

Оператор switch(){case:} в C:

Синтаксис:
switch ( /*Израз*/ ) 
{
    case /*Стойност1*/:
        //Код за изпълнение, ако изразът е равен на Стойност1
        break;
    case /*Стойност2*/:
        //Код за изпълнение, ако изразът е равен на Стойност2
        break;
    //Може да има повече case блокове
    default:
        //Код за изпълнение, ако изразът не съвпада с нито една от стойностите
}

Пример:
#include <stdio.h>
int main() {
    int day;
    printf("Въведете число от 1 до 7, представляващо ден от седмицата: ");
    scanf("%d", &day);
    
    switch (day) 
    {
        case 1:
            printf("Понеделник\n");
            break;
        case 2:
            printf("Вторник\n");
            break;
        case 3:
            printf("Сряда\n");
            break;
        case 4:
            printf("Четвъртък\n");
            break;
        case 5:
            printf("Петък\n");
            break;
        case 6:
            printf("Събота\n");
            break;
        case 7:
            printf("Неделя\n");
            break;
        default:
            printf("Невалиден ден от седмицата.\n");
    }
    
    return 0;
}

/////////////////////////////////////////////////////////////////////////////////////////////

Цикъл for(){} в C:

Синтаксис:
for ( /*Инициализация*/; /*Условие*/; /*Инкремент/Декремент*/ ) 
{
    //Код за изпълнение в цикъла
}

Пример:
#include <stdio.h>
int main() 
{
    int i;
    printf("Числата от 1 до 10 са:\n");
    
    for (i = 1; i <= 10; i++) 
    {
        printf("%d ", i);
    }
    
    printf("\n");
    return 0;
}

/////////////////////////////////////////////////////////////////////////////////////////////

Безкраен цикъл for(){} в C:

Синтаксис:
for (;;) 
{
    //Код за изпълнение в безкраен цикъл
}

Пример:
#include <stdio.h>
int main() 
{
    int count = 0;
    printf("Това е безкраен цикъл. Натиснете Ctrl+C за прекъсване.\n");
    
    for (;;) 
    {
        printf("Брой: %d\n", count);
        count++;
    }
    
    return 0;
}

/////////////////////////////////////////////////////////////////////////////////////////////

Повече от една актуализация в един цикъл for(){} в C:

Синтаксис:
for ( /*Инициализация 1*/, /*Инициализация 2*/; /*Условие 1*/, /*Условие 2*/; /*Инкремент/Декремент1*/, /*Инкремент/Декремент2*/ ) 
{
    //Код за изпълнение в цикъла
}   

Пример:
#include <stdio.h>
int main() 
{
    int i, j;
    printf("Таблица на умножение от 1 до 5:\n");
    
    for (i = 1, j = 1; i <= 5 && j <= 5; i++, j++) 
    {
        printf("%d ", i * j);
    }
    printf("\n");

    return 0;
}

/////////////////////////////////////////////////////////////////////////////////////////////

Цикъл while(){} в C:

Синтаксис:
while ( /*Условие*/ )
{
    //Код за изпълнение в цикъла
}

Пример:
#include <stdio.h>
int main()
{
    int count = 1;
    printf("Числата от 1 до 10 са:\n");
    
    while (count <= 10) 
    {
        printf("%d ", count);
        count++;
    }
    
    printf("\n");
    return 0;
}

/////////////////////////////////////////////////////////////////////////////////////////////

Безкраен цикъл while(){} в C:

Синтаксис:
while (1)
{
    //Код за изпълнение в безкраен цикъл

#include <stdio.h>

/////////////////////////////////////////////////////////////////////////////////////////////

Цикъл do{} while() в C:

Синтаксис:
do
{
    //Код за изпълнение в цикъла
} 
while ( /*Условие*/ );

Пример:
#include <stdio.h>
int main()
{
    int count = 1;
    printf("Числата от 1 до 10 са:\n");
    
    do 
    {
        printf("%d ", count);
        count++;
    } while (count <= 10);
    
    printf("\n");
    return 0;
}
